\begin{FSChapterPage}{7}{Sobre o Regulamento Geral}
\label{toc:7}
\FSSectionStart[toc:7.1]{7.1 Vigência}
\FSRuleA{7.1.1}{O presente Regulamento encontra-se em vigor desde o momento em que aprovado e apresentado.}
\FSRuleA{7.1.2}{O presente Regulamento é considerado caducado a partir do momento de entrada em vigor de Regulamento retificado.}

\FSSection[toc:7.2]{7.2 Revisão}
\FSRuleA{7.2.1}{O presente Regulamento deve ser revisto sempre que considerado pertinente.}
\FSRuleA{7.2.2}{Se retificado, o Regulamento Geral revisto deve ser proposto a aprovação.}

\FSSection[toc:7.3]{7.3 Casos Omissos}
\FSRuleA{7.3.1}{Qualquer eventualidade que não seja abrangida pelo presente Regulamento deve ser decidida pela Direção, segundo critérios de razoabilidade e funcionalidade.}

\FSSection[toc:7.4]{7.4 Protocolos}
\FSRuleA{7.4.1}{Extensões ao Regulamento Geral podem ser executadas pela Board - denominadas de Protocolos.}
\FSRuleA{7.4.2}{Os protocolos nunca devem isolada- ou coletivamente sobrepor-se ao presente Regulamento Geral, cuja autoridade é máxima.}
\end{FSChapterPage}
